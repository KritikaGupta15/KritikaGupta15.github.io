file.seekg(0, std::ios::end);
std::streamoff pos = file.tellg();
while (pos > 0) {
    std::size_t to_read = std::min(BUFFER_SIZE, pos);
    pos -= to_read;
    file.seekg(pos);
    file.read(buffer.data(), to_read);
